# =============================================================================
#  NOTES - Deploiement Odoo RBM-LOGISTIQUE sur Kubernetes
#  Chart : bitnami/odoo  --version 28.2.10
# =============================================================================
## -----------------------------------------------------------------------------
## 1. PRE-REQUIS
## -----------------------------------------------------------------------------
# StorageClass utilise : "local-path" (rancher.io/local-path)
#   -> defini dans ../postgres/local-path-provisioner.yaml
#   -> volumeBindingMode: WaitForFirstConsumer / reclaimPolicy: Delete
#
# Verifier que le provisioner et le StorageClass sont bien installes :
kubectl apply -f ../postgres/local-path-provisioner.yaml
kubectl get storageclass
# (on doit voir "local-path")
## -----------------------------------------------------------------------------
## 2. AJOUT DU DEPOT HELM BITNAMI
## -----------------------------------------------------------------------------
helm repo add bitnami https://charts.bitnami.com/bitnami
helm repo update
## -----------------------------------------------------------------------------
## 3. INSTALLATION / DEPLOIEMENT
## -----------------------------------------------------------------------------
# Creation du namespace dedie
kubectl create namespace rbm-logistique
# Installation avec le fichier de surcharge values.yml
helm install rbm-logis bitnami/odoo \
  --version 28.2.10 \
  --namespace rbm-logistique \
  -f values.yml
# Mise a jour ulterieure du release :
#   helm upgrade rbm-logis bitnami/odoo --version 28.2.10 -n rbm-logistique -f values.yml
## -----------------------------------------------------------------------------
## 4. CHOIX IMPORTANTS DU values.yml (overrides)
## -----------------------------------------------------------------------------
# - fullnameOverride: rbm-logistique      -> noms de ressources previsibles
# - service.type: NodePort (port 30069)   -> acces temporaire (Ingress plus tard)
# - persistence Odoo       : local-path, 10Gi, RWO, resourcePolicy=keep
# - persistence PostgreSQL : local-path, 8Gi,  RWO, resourcePolicy=keep
#     -> resourcePolicy=keep : les PVC survivent a un "helm uninstall"
#        (important pour la securite des donnees avant/apres backup Velero)
# - updateStrategy: Recreate -> obligatoire car volume RWO (local-path) non
#        montable sur 2 pods simultanement
# - global.security.allowInsecureImages: true -> compatibilite catalogue Bitnami
# - networkPolicy desactivee -> garantit l'acces via NodePort (a reactiver avec Ingress)
# - startupProbe tolerante   -> le 1er demarrage d'Odoo (init BD) est long
#
# >>> IMPORTANT - IMAGES BITNAMI (migration sept. 2025) <<<
# Les images "docker.io/bitnami/odoo" et "docker.io/bitnami/postgresql" ont ete
# retirees du Docker Hub (erreur ImagePullBackOff / "not found").
# On pointe donc vers le depot de maintien "bitnamilegacy" (memes tags) :
#   image.repository            : bitnamilegacy/odoo
#   postgresql.image.repository : bitnamilegacy/postgresql
# Verification de disponibilite d'un tag (exemple) :
#   docker manifest inspect docker.io/bitnamilegacy/postgresql:17.5.0-debian-12-r20
# NB: ces images ne recevront plus de mises a jour de securite. A terme,
#     migrer vers "Bitnami Secure Images" (bitnamisecure) ou une autre base.

## -----------------------------------------------------------------------------
## 7. BACKUP ULTERIEUR AVEC VELERO (rappel)
## -----------------------------------------------------------------------------
# Les PVC utilisent local-path (volumes "hostPath") -> activer File System Backup
# (anciennement Restic/Kopia) pour sauvegarder le contenu des volumes.
#
# Exemple de sauvegarde du namespace complet avec les volumes :
#   velero backup create rbm-logis-backup \
#     --include-namespaces rbm-logistique \
#     --default-volumes-to-fs-backup
#
# Restauration :
#   velero restore create --from-backup rbm-logis-backup
